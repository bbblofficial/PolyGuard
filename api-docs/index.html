کتابخانه‌ها

کتابخانه یا پکیج خاصی لازم نیست؛ فقط یک درخواست HTTP عادی به BASE_URL (لینک زنده از api/links.json) کافی است.

import requests

PAGES = "https://USER.github.io/REPO/api/links.json"
BASE = requests.get(PAGES, timeout=10).json()["base_url"]   # the live tunnel link
HEADERS = {}                                                # {"X-API-Key": "..."} if the server needs a key


def guard(path, **body):
    r = requests.post(BASE + "/api/v1/" + path, json=body, headers=HEADERS, timeout=60)
    if r.status_code == 429:                                 # rate limited
        raise RuntimeError(f"retry in {{r.headers['Retry-After']}}s")
    r.raise_for_status()
    return r.json()


guard("check", text="to khar hasti")["decision"]           # "TOXIC"
guard("verdict", text="Have a nice day")["flagged"]        # False
guard("censor", text="you are a k1r")["text"]             # "you are a ***"
guard("check/batch", texts=["salam", "khafe sho"])["summary"]
const PAGES = "https://USER.github.io/REPO/api/links.json";
const { base_url: BASE } = await (await fetch(PAGES)).json();   // the live tunnel link

async function guard(path, body) {
  const r = await fetch(`${BASE}/api/v1/${path}`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  return r.json();
}

(await guard("check", { text: "to khar hasti" })).decision;   // "TOXIC"
